cmake_minimum_required(VERSION 3.16)
project(bs0013_midterm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(registry src/cargo_registry.cpp)
target_include_directories(registry PUBLIC include)
target_compile_options(registry PRIVATE -Wall -Wextra -Wpedantic)
option(ENABLE_SANITIZERS "Enable address and undefined-behavior checks" OFF)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(registry PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
endif()
enable_testing()
add_executable(public_tests tests/public_tests.cpp)
target_link_libraries(public_tests PRIVATE registry)
add_test(NAME public_tests COMMAND public_tests)
add_executable(student_tests tests/student_tests.cpp)
target_link_libraries(student_tests PRIVATE registry)
add_test(NAME student_tests COMMAND student_tests)
